function renderStep(stepKey) {
    const container = document.getElementById('game-container');
    const data = steps[stepKey];

    let optionsHtml = '';
    data.options.forEach(opt => {
        if (opt.disabled) {
            optionsHtml += `<button disabled>${opt.text}</button>`;
        } else {
            optionsHtml += `<button onclick="renderStep('${opt.next}')">${opt.text}</button>`;
        }
    });

    container.innerHTML = `
        <div class="window">
            <div class="title-bar">
                <div class="title-bar-text">${data.title}</div>
            </div>
            <div class="window-body">
                <p>${data.text}</p>
                <div class="options-container">
                    ${optionsHtml}
                </div>
            </div>
        </div>
    `;
}
